cmake_minimum_required(VERSION 3.22)
project(ios27_android C)
set(IOS27_BUILD_TESTS OFF CACHE BOOL "" FORCE)
get_filename_component(IOS27_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
add_subdirectory("${IOS27_ROOT}" "${CMAKE_BINARY_DIR}/runtime")
add_executable(ios27_android_runner runner.c)
target_link_libraries(ios27_android_runner PRIVATE ios27_core)
target_compile_options(ios27_android_runner PRIVATE -Wall -Wextra -Werror)
add_executable(ios27_android_loader "${IOS27_ROOT}/src/loader.c")
target_compile_definitions(ios27_android_loader PRIVATE IOS27_WITH_SERVICES=1)
target_link_libraries(ios27_android_loader PRIVATE ios27_core ios27_service_bridge)
foreach(target ios27_android_runner ios27_android_loader)
    set_target_properties(${target} PROPERTIES PREFIX "lib" SUFFIX ".so" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_LIBRARY_OUTPUT_DIRECTORY}")
    target_link_options(${target} PRIVATE "LINKER:-z,now" "LINKER:-z,max-page-size=16384")
endforeach()
